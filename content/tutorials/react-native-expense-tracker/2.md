## 2.3 `hooks/useExpenses.js` — Data Layer (State + Hooks + AsyncStorage)

```jsx
import { useState, useEffect, useCallback } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

const STORAGE_KEY = 'expenses';

export function useExpenses() {
  const [expenses, setExpenses] = useState([]);
  const [loading, setLoading] = useState(true);

  // App စဖွင့်ချိန် saved data load (useEffect)
  useEffect(() => {
    AsyncStorage.getItem(STORAGE_KEY).then((data) => {
      if (data) setExpenses(JSON.parse(data));
      setLoading(false);
    });
  }, []);

  // expenses ပြောင်းတိုင်း AsyncStorage ထဲ ပြန်သိမ်း
  useEffect(() => {
    if (!loading) {
      AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(expenses));
    }
  }, [expenses, loading]);

  const addExpense = useCallback((title, amount) => {
    const newExpense = {
      id: Date.now().toString(),
      title,
      amount: parseFloat(amount),
    };
    setExpenses((prev) => [newExpense, ...prev]); // state mutate မလုပ်ဘဲ array အသစ်ဆောက်
  }, []);

  const deleteExpense = useCallback((id) => {
    setExpenses((prev) => prev.filter((e) => e.id !== id));
  }, []);

  const total = expenses.reduce((sum, e) => sum + e.amount, 0);

  return { expenses, addExpense, deleteExpense, total, loading };
}
```

### (c) `components/MonthSummary.jsx` — Date Picker + Selected Month Total

```jsx
import { useState } from 'react';
import { View, Text, Pressable, StyleSheet } from 'react-native';
import DateTimePicker from '@react-native-community/datetimepicker';

export default function MonthSummary({ selectedDate, onChangeDate, total }) {
  const [showPicker, setShowPicker] = useState(false);

  function handleChange(event, date) {
    setShowPicker(false);
    if (date) onChangeDate(date);
  }

  const monthLabel = selectedDate.toLocaleString('default', {
    month: 'long',
    year: 'numeric',
  });

  return (
    <View style={styles.container}>
      <Pressable onPress={() => setShowPicker(true)}>
        <Text style={styles.month}>{monthLabel} ▾</Text>
      </Pressable>
      <Text style={styles.total}>{total.toLocaleString()} Ks</Text>

      {showPicker && (
        <DateTimePicker
          value={selectedDate}
          mode="date"
          display="spinner"
          onChange={handleChange}
        />
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { padding: 20, backgroundColor: '#2c3e50', borderRadius: 12, margin: 16 },
  month: { color: '#bdc3c7', fontSize: 14, marginBottom: 4 },
  total: { color: '#fff', fontSize: 32, fontWeight: 'bold' },
});
```

> `mode="date"` နဲ့ day/month/year အားလုံးပြပေမယ့် **month + year ကိုပဲ** သုံးမှာမို့ ရွေးလိုက်တဲ့ date ရဲ့ day က ဂရုမစိုက်ပါဘူး — code ထဲမှာ `getMonth()`/`getFullYear()` ကိုပဲ compare လုပ်မှာပါ (အောက်မှာ ရှင်းထားပါတယ်)။

### (d) `app/index.jsx` — Category Filter + Monthly Summary ချိတ်ဆက်ခြင်း (ပြင်ဆင်ပြီးသား full version)

```jsx
import { useState } from 'react';
import { View, FlatList, Text, Pressable, StyleSheet } from 'react-native';
import { Link } from 'expo-router';
import { useExpenses } from '../hooks/useExpenses';
import ExpenseCard from '../components/ExpenseCard';
import CategoryFilter from '../components/CategoryFilter';
import MonthSummary from '../components/MonthSummary';

function isSameMonth(isoDate, refDate) {
  const d = new Date(isoDate);
  return d.getMonth() === refDate.getMonth() && d.getFullYear() === refDate.getFullYear();
}

export default function Home() {
  const { expenses, deleteExpense, loading } = useExpenses();
  const [selectedCategory, setSelectedCategory] = useState('All');
  const [selectedDate, setSelectedDate] = useState(new Date());

  if (loading) return null;

  // 1️⃣ လအလိုက် filter
  const monthlyExpenses = expenses.filter((e) => isSameMonth(e.date, selectedDate));

  // 2️⃣ category အလိုက် filter (All ဆိုရင် အားလုံးပြ)
  const filteredExpenses = monthlyExpenses.filter((e) =>
    selectedCategory === 'All' ? true : e.category === selectedCategory
  );

  // ဒီလ၊ ဒီ category အတွက် total
  const total = filteredExpenses.reduce((sum, e) => sum + e.amount, 0);

  return (
    <View style={styles.container}>
      <MonthSummary
        selectedDate={selectedDate}
        onChangeDate={setSelectedDate}
        total={total}
      />

      <CategoryFilter selected={selectedCategory} onSelect={setSelectedCategory} />

      <FlatList
        data={filteredExpenses}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <ExpenseCard
            title={item.title}
            amount={item.amount}
            category={item.category}
            onDelete={() => deleteExpense(item.id)}
          />
        )}
        ListEmptyComponent={<Text style={styles.empty}>No expenses for this filter</Text>}
      />

      <Link href="/add-expense" asChild>
        <Pressable style={styles.fab}>
          <Text style={styles.fabText}>+</Text>
        </Pressable>
      </Link>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#f4f4f4' },
  empty: { textAlign: 'center', color: '#999', marginTop: 40 },
  fab: {
    position: 'absolute', right: 24, bottom: 24, width: 56, height: 56,
    borderRadius: 28, backgroundColor: '#2c3e50',
    justifyContent: 'center', alignItems: 'center',
  },
  fabText: { color: '#fff', fontSize: 28, lineHeight: 28 },
});
```

### ရလဒ်

Home screen ကို ဖွင့်လိုက်ရင်:
- အပေါ်ဆုံးမှာ **လ (month) ရွေးလို့ရတဲ့ summary card** — နှိပ်ရင် date picker ပေါ်ပြီး လပြောင်းနိုင်၊ ရွေးထားတဲ့လရဲ့ total ကို ပြသ
- အဲ့အောက်မှာ **category chip များ** (All / Food / Transport / Bills / Other) — နှိပ်ရင် list ကို instant filter လုပ်ပေး
- List ထဲက card တစ်ခုချင်းစီမှာ category label ထည့်ပြထားတယ်

Filter ၂ ခုလုံးက **state** ချည်းသာဖြစ်ပြီး (`selectedCategory`, `selectedDate`), `expenses` array တစ်ခုတည်းကနေ `.filter()` ချည်းနဲ့ လုပ်ထားတာမို့ **extra data fetch/query မလိုပါဘူး** — Part 1 က state/props concept တွေကိုပဲ ထပ်ကာထပ်ကာ သုံးထားခြင်းသာ ဖြစ်ပါတယ်။

### ဆက်တိုးနိုင်တဲ့ Feature များ

- Chart library (`victory-native`, `react-native-svg-charts`) နဲ့ category-wise spending graph ပြခြင်း
- Category အလိုက် budget limit သတ်မှတ်ပြီး limit ကျော်ရင် warning ပြခြင်း
